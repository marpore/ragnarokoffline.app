// Outbound ferry: talk to the NPC only (no walk-over warp — that duplicated
// the same trip and felt redundant next to Osric).
prontera,155,178,4	script	Ferryman Osric#ferry	4_M_SEAMAN,{
	mes "[Ferryman Osric]";
	mes "Boat leaves for the island whenever you like. No charge.";
	next;
	if (select("Take me across:Maybe later") == 2) {
		mes "[Ferryman Osric]";
		mes "She'll be here.";
		close;
	}
	mes "[Ferryman Osric]";
	mes "This trip leaves your savepoint unchanged. The island ferryman can set it back to Prontera.";
	close2;
	// close2 first so the dialogue box is gone before the map changes.
	// Do not savepoint on ro_isle.
	warp "ro_isle", 40, 40;
	end;
}
